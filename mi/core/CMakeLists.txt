set(HEADER_FILES
    include/core/base.h
    include/core/common.h
    include/core/infra.h
    include/core/meta.h
    include/core/pixel_format.h
    include/core/platform.h
    include/core/refcounted.h
    include/core/thr.h

    include/core/util/queue.h
    include/core/util/segment_allocator.h
    include/core/util/slot_allocator.h
    include/core/util/byte_strided_span.h
)

set(SOURCE_FILES
    blobres.cpp
    task.cpp
    thr.cpp
        include/core/fwd.h
)

add_library(core STATIC ${HEADER_FILES} ${SOURCE_FILES})
target_link_libraries(core PUBLIC glm::glm)
target_include_directories(core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

# Output an identity file to specify the root path of the project for executables
set(IDENTITY_FILE "${CMAKE_BINARY_DIR}/mi_renderer_identity")
message(STATUS "Identity file: ${IDENTITY_FILE}")
add_custom_command(
        OUTPUT ${IDENTITY_FILE}
        COMMAND ${CMAKE_COMMAND} -E echo "This autogenerated file marks the project root for compiled executables." > ${IDENTITY_FILE}
        COMMAND ${CMAKE_COMMAND} -E echo "Do not remove it." >> ${IDENTITY_FILE}
        COMMENT "This file marks the project root for compiled executables."
        VERBATIM
)
add_custom_target(generate_identity_file ALL DEPENDS ${IDENTITY_FILE})
add_dependencies(core generate_identity_file)
